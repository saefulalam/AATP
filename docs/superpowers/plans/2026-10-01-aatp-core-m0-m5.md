> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Python reference implementation of AATP v0.2 Core (M0–M5) that reproduces test vectors V01–V09 byte-exact and passes negative tests N01–N16, per `AATP_v0_2_Blueprint.md`.

**Architecture:** Modular package `aatp/` mirroring blueprint §18.1: canon → dsse/signer → ids/card → artifact → node → chunks/seal → transfer. Signer/KeyStore isolates all private keys (SIG-001). Fail-closed everywhere; gaps go to `DECISIONS.md` (blueprint §0.4).

**Tech Stack:** Python 3.12 (3.11+ OK), `cryptography` (Ed25519/X25519/ChaCha20-Poly1305/HKDF-SHA256), `pyhpke` (HPKE KEM 0x0020/KDF 0x0001/AEAD 0x0003 — validated against RFC 9180 vectors in Task 0), `pytest`.

**Spec:** `docs/superpowers/specs/2026-10-01-aatp-core-m0-m5-design.md`
**Blueprint:** `AATP_v0_2_Blueprint.md`

## Global Constraints

- One rule ID = one test, named after the rule ID (e.g. `test_GEN010_reject_float`), failing-first (§18.4.1).
- Never log or print private keys, CEKs, or plaintext artifact bytes (§18.4.2).
- Fail closed on ambiguity; record in `DECISIONS.md` (§0.4, §18.4.3).
- No features/extensions/algorithms outside the blueprint (§18.4.4).
- Vectors are regenerated by `reference_vectors.py`; V01–V09 must match byte-for-byte (§18.4.5).
- Only SHA-256; digests are `sha256:<64 lowercase hex>` (GEN-002).
- Timestamps exactly `YYYY-MM-DDTHH:MM:SSZ` UTC (GEN-004).
- Namespace prefix `aatp`; capability IDs `aatp.<area>.<name>` (GEN-005).
- IDs: `msg_`/`tx_`/`task_` + UUIDv7 lowercase hyphenated, or opaque ≤128 chars `[A-Za-z0-9_.:-]` (GEN-006).
- `cryptography` pinned ≥43; `pyhpke` pinned 0.6.5; `pytest` ≥8.
- Domain-separation constants are byte-exact ASCII with trailing `\n` (§3.3):
  `D_CONTENT="AATP/0.2/CONTENT\n"`, `D_NODE="AATP/0.2/NODE\n"`, `D_AGENT="AATP/0.2/AGENT-ID\n"`, `D_CHUNK="AATP/0.2/CHUNK\n"`, `D_PACKET="AATP/0.2/PACKET\n"`, `D_CEKWRAP="AATP/0.2/CEK-WRAP\n"`, `D_GRANT="AATP/0.2/GRANT\n"` (§10.1; defined now, used M6).
- Default limits exist and are configurable (§2.1): `max_message_bytes=262144`, `max_manifest_entries=10000`, `max_path_bytes=1024`, `max_total_artifact_bytes=4294967296`, `chunk_size default=1048576 max=16777216`, `total_chunks max=4294967295`, `send_window_chunks=16`, `clock_skew_seconds=120`, `message_lifetime_max_seconds=86400`, `offer_lifetime_default_seconds=3600`.
- Errors (§14): `AATPError(code, numeric, detail=None)`; numeric map per blueprint §14.
- Git: init repo in Task 0; commit per task with message `feat(Mx): <rule-scope>`.

# tests/test_hpke_spike.py
import json, base64
from pathlib import Path

def test_pyhpke_rfc9180_A1_1_roundtrip():
    from pyhpke import CipherSuite, KEMId, KDFId, AEADId
    vec_path = Path(__file__).parent / "vectors" / "rfc9180_hpke.json"
    v = [x for x in json.loads(vec_path.read_text())
         if x["mode"] == 0 and x["kem_id"] == 0x0020 and x["kdf_id"] == 0x0001 and x["aead_id"] == 0x0003][0]
    suite = CipherSuite.new(KEMId.DHKEM_X25519_HKDF_SHA256, KDFId.HKDF_SHA256, AEADId.ChaCha20Poly1305)
    skR = suite.kem.deserialize_private_key(bytes.fromhex(v["skRm"]))
    pkR = suite.kem.deserialize_public_key(bytes.fromhex(v["pkRm"]))
    info = bytes.fromhex(v["info"])
    enc = bytes.fromhex(v["enc"])
    # Gate 1: official vector ciphertexts open to official plaintexts (seq 0 = single-shot)
    e0 = v["encryptions"][0]
    pt = suite.open(enc, skR, info, bytes.fromhex(e0["aad"]), bytes.fromhex(e0["ct"]))
    assert pt == bytes.fromhex(e0["pt"])
    # Gate 2: fresh seal/open round-trip
    enc2, ct2 = suite.seal(pkR, b"roundtrip", info, b"")
    assert suite.open(enc2, skR, info, b"", ct2) == b"roundtrip"
```

Note: a random-ephemeral `seal` can never reproduce the vector's `enc`; only `open` of the vector `ct` is a valid known-answer test. Argument order of `seal/open` must be taken from pyhpke's actual API (inspect signatures; do not guess). The gate is: RFC 9180 base-mode X25519/HKDF-SHA256/ChaCha20-Poly1305 vector ciphertext opens to the vector plaintext, and round-trip works. # aatp/canon.py
import json
from aatp.errors import err

MAX_DEPTH = 16

def _check_depth(obj, d=0):
    if d > MAX_DEPTH:
        raise err("ERR_MALFORMED_MESSAGE", "nesting depth > 16")
    if isinstance(obj, dict):
        for v in obj.values():
            _check_depth(v, d + 1)
    elif isinstance(obj, list):
        for v in obj:
            _check_depth(v, d + 1)

def _validate(obj):
    if isinstance(obj, float):
        raise err("ERR_MALFORMED_MESSAGE", "floats forbidden in canonical JSON")
    if isinstance(obj, bool) or obj is None or isinstance(obj, int):
        if isinstance(obj, int) and not isinstance(obj, bool) and abs(obj) > 2**53 - 1:
            raise err("ERR_MALFORMED_MESSAGE", "integer exceeds 2^53-1")
        return
    if isinstance(obj, str):
        return
    if isinstance(obj, list):
        for v in obj: _validate(v)
        return
    if isinstance(obj, dict):
        for k, v in obj.items():
            if not k.isascii():
                raise err("ERR_MALFORMED_MESSAGE", "non-ASCII object key")
            _validate(v)
        return
    raise err("ERR_MALFORMED_MESSAGE", f"unsupported type {type(obj)}")

def jcs(obj) -> bytes:
    _check_depth(obj)
    _validate(obj)
    return json.dumps(obj, ensure_ascii=False, separators=(",", ":"),
                      sort_keys=True).encode("utf-8")

def parse_jcs(data: bytes):
    # Reject duplicates via object_pairs_hook
    def no_dup(pairs):
        seen, out = set(), {}
        for k, v in pairs:
            if k in seen:
                raise err("ERR_MALFORMED_MESSAGE", "duplicate key")
            seen.add(k)
            out[k] = v
        return out
    text = data.decode("utf-8")
    obj = json.loads(text, object_pairs_hook=no_dup,
                     parse_float=lambda s: (_ for _ in ()).throw(
                         err("ERR_MALFORMED_MESSAGE", "float forbidden")))
    _check_depth(obj)
    _validate(obj)
    if jcs(obj) != data:
        raise err("ERR_MALFORMED_MESSAGE", "not in canonical form")
    return obj
```

# aatp/signer.py
from cryptography.hazmat.primitives.asymmetric.ed25519 import Ed25519PrivateKey
from cryptography.hazmat.primitives import serialization

class SoftwareSigner:
    """SIG-001: key never leaves this object; only sign() is exposed."""
    def __init__(self, sk: Ed25519PrivateKey, key_id: str):
        self._sk = sk  # never expose
        self._key_id = key_id

    @classmethod
    def from_seed(cls, seed32: bytes, key_id: str) -> "SoftwareSigner":
        return cls(Ed25519PrivateKey.from_private_bytes(seed32), key_id)

    def key_id(self) -> str:
        return self._key_id

    def public_key_raw(self) -> bytes:
        return self._sk.public_key().public_bytes(
            serialization.Encoding.Raw, serialization.PublicFormat.Raw)

    def sign(self, payload_type: str, payload: bytes) -> bytes:
        # CRY-002: sign PAE bytes directly, no pre-hash
        from aatp.dsse import pae
        return self._sk.sign(pae(payload_type, payload))
```

# aatp/dsse.py
import base64
from cryptography.hazmat.primitives.asymmetric.ed25519 import Ed25519PublicKey
from cryptography.exceptions import InvalidSignature
from aatp.canon import jcs, parse_jcs
from aatp.errors import err

def b64u(b: bytes) -> str:
    return base64.urlsafe_b64encode(b).rstrip(b"=").decode()

def b64u_dec(s: str) -> bytes:
    pad = "=" * (-len(s) % 4)
    return base64.urlsafe_b64decode(s + pad)

def pae(payload_type: str, payload: bytes) -> bytes:
    t = payload_type.encode("utf-8")
    return (b"DSSEv1 " + str(len(t)).encode() + b" " + t + b" " +
            str(len(payload)).encode() + b" " + payload)

def make_envelope(payload_type: str, payload: bytes, signer) -> dict:
    sig = signer.sign(payload_type, payload)
    return {"payloadType": payload_type, "payload": b64u(payload),
            "signatures": [{"keyid": signer.key_id(), "sig": b64u(sig)}]}

def verify_envelope(env: dict, expected_payload_type: str, public_key_raw: bytes) -> dict:
    if env.get("payloadType") != expected_payload_type:
        raise err("ERR_MALFORMED_MESSAGE", "payloadType mismatch")  # SIG-003
    payload = b64u_dec(env["payload"])
    obj = parse_jcs(payload)          # SIG-002: canonical check
    if jcs(obj) != payload:
        raise err("ERR_MALFORMED_MESSAGE", "re-canonicalization differs")
    sig = b64u_dec(env["signatures"][0]["sig"])
    try:
        Ed25519PublicKey.from_public_bytes(public_key_raw).verify(sig, pae(expected_payload_type, payload))
    except InvalidSignature:
        raise err("ERR_SIGNATURE_INVALID")
    return obj
```

Test `test_CRY003` asserts rejection; if the library accepts a malleable sig, record strict-mode choice in `DECISIONS.md` and add explicit S<L check. CRY-006: all randomness via `secrets`.

D_AGENT = b"AATP/0.2/AGENT-ID\n"
